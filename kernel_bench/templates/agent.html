import requests
import time

BENCH_URL = "http://localhost:8080"

# 1. Connect to Agent Home Page & Index
index = requests.get(f"{BENCH_URL}/api/agent").json()
print("Mission:", index["mission"]["objective"])
print("Current Best pp512:", index["state"]["best_pp512_tps"], "t/s")
print("Current Best tg128:", index["state"]["best_tg128_tps"], "t/s")

# 2. Inspect Bottlenecks
bottlenecks = requests.get(f"{BENCH_URL}/api/analyze", params={"filter": "fallback"}).json()
print("Top fallback:", bottlenecks.get("cpu_fallbacks", [{}])[0].get("op"))

# 3. Submit Optimization Kernel
payload = {
    "idea": "Broadcast Multiply CUDA kernel to eliminate REF_MUL",
    "filename": "kernels/cuda/mul/NC_F32_ND.cu",
    "source": "/* your CUDA code with REGISTER_KERNEL */",
    "backend": "cuda"
}
job = requests.post(f"{BENCH_URL}/api/iteration/submit", json=payload).json()
job_id = job["job_id"]
print(f"Queued Job {job_id} for Version {job['version']}. Polling...")

# 4. Poll until completed
while True:
    status = requests.get(f"{BENCH_URL}/api/jobs/{job_id}").json()
    print("Step:", status.get("step"), "Status:", status.get("status"))
    if status.get("status") in ("completed", "failed"):
        print("Final result:", status.get("metrics"))
        break
    time.sleep(5)
